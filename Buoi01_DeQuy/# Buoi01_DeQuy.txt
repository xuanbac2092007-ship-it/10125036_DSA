using System;

namespace BTTH1
{
    static class B5_Fibo
    {
        static int FiboLap(int n)
        {
            if (n == 0) return 0;
            if (n == 1) return 1;

            int a = 0;
            int b = 1;
            int c = 0;

            for (int i = 2; i <= n; i++)
            {
                c = a + b;
                a = b;
                b = c;
            }

            return c;
        }
        static int FiboDeQuy(int n)
        {
            if (n == 0) return 0;
            if (n == 1) return 1;

            return FiboDeQuy(n - 1) + FiboDeQuy(n - 2);
        }

        public static void Run()
        {
            Console.Write("Nhap n: ");
            int n = int.Parse(Console.ReadLine());

            if (n < 0 || n > 40)
            {
                Console.WriteLine("Vui long nhap n tu 0 den 40!");
                return;
            }
Console.WriteLine("Fibonacci bang vong lap: " + FiboLap(n));
            Console.WriteLine("Fibonacci bang de quy: " + FiboDeQuy(n));
        }
    }
}
```

namespace BTTH1
{
    static class B2C2
    {
        public static void Run()
        {
            int n = hamhotro.DocSo("Nhap so phan tu", 10);

            if (n <= 0) return;

            int[] a = hamhotro.TaoMangNgauNhien(n, 100);

            for (int i = 0; i < a.Length - 1; i++)
            {
                for (int j = 0; j < a.Length - 1 - i; j++)
                {
                    if (a[j] > a[j + 1])
                        hamhotro.Swap(a, j, j + 1);
                }
            }

            Console.WriteLine("Mang sau sap xep:");
            hamhotro.InMang(a);

            int x = hamhotro.DocSo("Nhap so can tim", 5);

            int dau = 0;
            int cuoi = n - 1;
            int vt = -1;

            while (dau <= cuoi)
            {
                int giua = (dau + cuoi) / 2;

                if (a[giua] == x)
                {
                    vt = giua;
                    break;
                }
                else if (a[giua] < x)
                    dau = giua + 1;
                else
                    cuoi = giua - 1;
            }

            if (vt == -1)
                Console.WriteLine("Khong tim thay!");
            else
                Console.WriteLine("Tim thay tai vi tri: " + vt);
        }
    }
}
```

        {
            if (i == n)
